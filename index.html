Petunjuk & Security

🔐 Firestore Security Rules (production):

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Aplikasi publik: hanya boleh CREATE
    match /reports/{doc}           { allow create: if true; allow read, update, delete: if request.auth.token.admin == true; }
    match /layanan_requests/{doc}  { allow create: if true; allow read, update, delete: if request.auth.token.admin == true; }
    match /partisipasi/{doc}       { allow create: if true; allow read, update, delete: if request.auth.token.admin == true; }
    match /users/{email}           { allow create, update: if true; allow read: if request.auth.token.admin == true; }
    match /otp_logs/{doc}          { allow create: if true; allow read: if request.auth.token.admin == true; }
    match /activity_logs/{doc}     { allow create: if true; allow read: if request.auth.token.admin == true; }
  }
}

⚠️ Untuk testing, sementara gunakan: allow read, write: if true; — jangan dipakai di production.
